FearStage.cs
/// <summary>
/// Represents the discrete fear stages used by the fear system.
/// The stages increase in intensity from <see cref="None"/> to <see cref="Max"/>.
/// </summary>
/// <remarks>
/// Stored as a <c>byte</c> to keep the serialized size small and stable.
/// The numeric values explicitly map increasing intensity to increasing numbers.
/// </remarks>
public enum FearStage : byte
{
    /// <summary>No fear / default state.</summary>
    None = 1,

    /// <summary>Low level of fear.</summary>
    Low = 2,

    /// <summary>Medium level of fear.</summary>
    Mid = 3,

    /// <summary>Maximum level of fear.</summary>
    Max = 4
}
IFearStageReceiver.cs
/// <summary>
/// Implemented by components that receive and apply a <see cref="FearStage"/> change.
/// The fear system calls <see cref="ApplyEffectsForStage(FearStage)"/> when a new
/// stage should be applied to the receiver (for example: vignette, audio, haptics).
/// </summary>
public interface IFearStageReceiver
{
	/// <summary>
	/// Apply the effects associated with the provided fear <paramref name="stage"/>.
	/// </summary>
	/// <param name="stage">The fear stage to apply.</param>
			void ApplyEffectsForStage(FearStage stage);
}
FearStageProvider.cs
using System.Collections.Generic;
using UnityEngine;

/// <summary>
/// Central hub for the fear system.
/// Holds the current <see cref="FearStage"/> and broadcasts stage changes to all
/// registered <see cref="IFearStageReceiver"/> components.
/// </summary>
[DisallowMultipleComponent]
public sealed class FearStageProvider : MonoBehaviour
{
	/// <summary>
	///    Initial fear stage applied on startup and exposed as the current provider state.
	///</summary>
	[Header("Initial stage")] [SerializeField]
	private FearStage _stage = FearStage.None;


	//private readonly HashSet<IFearStageReceiver> _registeredReceivers = new();

	private readonly List<IFearStageReceiver> _receivers = new();

	/// <summary>
	///    Global access to the active fear stage provider in the current scene.
	/// </summary>
	public static FearStageProvider Instance { get; private set; }

	/// <summary>The currently active fear stage.</summary>
	public FearStage Stage => _stage;

	/// <summary>
	/// Initializes the singleton instance for this scene.
	/// Destroys this object if another provider instance already exists.
	/// </summary>
	private void Awake()
	{
		if (Instance != null && Instance != this)
		{
			Destroy(gameObject);
			return;
		}

		Instance = this;

		DontDestroyOnLoad(gameObject);

	}

	/// <summary>
	/// Cleans up the singleton instance reference if this provider is destroyed.
	/// </summary>
	private void OnDestroy()
	{
		if (Instance == this)
			Instance = null;
	}

	/// <summary>
	/// Registers a receiver and immediately applies the current stage so it stays in sync.
	/// </summary>
	/// <param name="receiver">Receiver instance to register.</param>
                    public void Register(IFearStageReceiver receiver)
	{
		if (receiver == null)
		{
			Debug.LogWarning("[FearStageProvider] Attempted to register a null receiver.", this);
			return;
		}

		_receivers.Add(receiver);
		receiver.ApplyEffectsForStage(_stage);
	}

	/// <summary>
	/// Unregisters a receiver so it no longer receives stage updates.
	/// </summary>
	/// <param name="receiver">Receiver instance to unregister.</param>
                    public void Unregister(IFearStageReceiver receiver)
	{
		if (receiver == null)
		{
			Debug.LogWarning("[FearStageProvider] Attempted to unregister a null receiver.", this);
			return;
		}

		_receivers.Remove(receiver);
	}

	/// <summary>
	/// Sets the fear stage to <paramref name="newStage"/> and notifies all receivers.
	/// Does nothing if <paramref name="newStage"/> is already active.
	/// </summary>
	/// <param name="newStage">The stage to transition to.</param>
                    public void SetStage(FearStage newStage)
	{
		// Skip if the new stage is the same as the current one to avoid redundant updates.
		if (_stage == newStage)
			return;

		_stage = newStage;
		ApplyToReceivers(_stage);
	}

	/// <summary>
	/// Advances the fear stage by one step (for example None to Low to Mid to Max).
	/// Does nothing when already at <see cref="FearStage.Max"/>.
	/// </summary>
	public void IncreaseStage()
	{
		if (_stage < FearStage.Max)
			SetStage((FearStage)((byte)_stage + 1));
		else
			Debug.LogWarning($"[FearStageProvider] Cannot increase stage — already at maximum ({_stage}).", this);
	}

	/// <summary>
	/// Reduces the fear stage by one step (for example Max to Mid to Low to None).
	/// Does nothing when already at <see cref="FearStage.None"/>.
	/// </summary>
	public void DecreaseStage()
	{
		if (_stage > FearStage.None)
			SetStage((FearStage)((byte)_stage - 1));
		else
			Debug.LogWarning($"[FearStageProvider] Cannot decrease stage — already at minimum ({_stage}).", this);
	}

	/// <summary>
	/// Forwards <paramref name="stage"/> to all currently registered receivers.
	/// </summary>
	/// <param name="stage">Fear stage to apply.</param>
                    private void ApplyToReceivers(FearStage stage)
	{
		// Notify all valid receivers of the stage change.
		foreach (IFearStageReceiver receiver in _receivers)
			receiver.ApplyEffectsForStage(stage);
	}
}
